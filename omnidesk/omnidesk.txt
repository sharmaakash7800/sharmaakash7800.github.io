    // Update Item Counts
    function updateCounts() {
      const items = getAllItems();
      const counts = {
        all: items.length,
        favorites: starredIds.length,
        dashboards: 0,
        sheets: 0,
        docs: 0,
        apps: 0,
        tools: 0
      };

      items.forEach(item => {
        if (counts[item.category] !== undefined) {
          counts[item.category]++;
        }
      });

      document.getElementById("countAll").textContent = counts.all;
      document.getElementById("countFavorites").textContent = counts.favorites;
      document.getElementById("countDashboards").textContent = counts.dashboards;
      document.getElementById("countSheets").textContent = counts.sheets;
      document.getElementById("countDocs").textContent = counts.docs;
      document.getElementById("countApps").textContent = counts.apps;
      document.getElementById("countTools").textContent = counts.tools;

      renderDrawerNav(counts);
    }

    // Render Mobile Drawer Items
    function renderDrawerNav(counts) {
      const keys = ["all", "favorites", "dashboards", "sheets", "docs", "apps", "tools"];
      drawerNavList.innerHTML = "";

      keys.forEach(key => {
        const isActive = currentCategory === key;
        const icon = key === "all" ? "✨" : key === "favorites" ? "⭐" : categoryConfig[key]?.icon || "🏷️";
        const navHtml = `
          <li class="drawer-nav-item ${isActive ? 'active' : ''}" onclick="selectCategoryFromDrawer('${key}')">
            <span>${icon} ${categoryNames[key]}</span>
            <span class="nav-count">${counts[key] || 0}</span>
          </li>
        `;
        drawerNavList.insertAdjacentHTML("beforeend", navHtml);
      });
    }

    // Toggle Favorite
    function toggleStar(itemId, event) {
      if (event) {
        event.preventDefault();
        event.stopPropagation();
      }

      if (starredIds.includes(itemId)) {
        starredIds = starredIds.filter(id => id !== itemId);
        showToast("Removed from Starred");
      } else {
        starredIds.push(itemId);
        showToast("⭐ Added to Starred!");
      }

      localStorage.setItem("omnidesk_starred", JSON.stringify(starredIds));
      updateCounts();
      renderItems();
    }

      if (event) {
        event.preventDefault();
        event.stopPropagation();
      }

      if (confirm("Are you sure you want to delete this custom link?")) {
        customLinks = customLinks.filter(item => item.id !== itemId);
        localStorage.setItem("omnidesk_custom_links", JSON.stringify(customLinks));
        showToast("🗑️ Link deleted!");
        updateCounts();
        renderItems();
      }
    }

    // Copy Link URL
    function copyUrl(url, event) {
      if (event) {
        event.preventDefault();
        event.stopPropagation();
      }

      navigator.clipboard.writeText(url).then(() => {
        showToast("📋 Link copied to clipboard!");
      }).catch(() => {
        showToast("Failed to copy link");
      });
    }

    // Toast
    function showToast(msg) {
      const toast = document.getElementById("toast");
      document.getElementById("toastMsg").textContent = msg;
      toast.classList.add("show");
      setTimeout(() => {
        toast.classList.remove("show");
      }, 2500);
    }

    function formatUrlPreview(url) {
      try {
        const parsed = new URL(url);
        return parsed.hostname.replace("www.", "");
      } catch (e) {
        return url;
      }
    }

    // Single Card HTML Generator
    function createCardHtml(item) {
      const isStarred = starredIds.includes(item.id);
      const isCustom = item.isCustom;

      return `
        <a class="card-item" href="${item.url}" target="_blank" rel="noopener noreferrer">
          <div class="card-main-info">
            <div class="card-icon">${item.icon || '🔗'}</div>
            <div class="card-text">
              <div class="card-title">${item.title}</div>
              <div class="card-url-preview">🔗 ${formatUrlPreview(item.url)}</div>
              <div class="card-tag-row">
                <span class="card-tag">${item.category}</span>
                ${isCustom ? '<span class="custom-badge">CUSTOM</span>' : ''}
              </div>
            </div>
          </div>

          <div class="card-actions">
            <button class="action-btn star-btn ${isStarred ? 'starred' : ''}" 
              title="${isStarred ? 'Unstar' : 'Star'}" 
              onclick="toggleStar('${item.id}', event)">
              ${isStarred ? '★' : '☆'}
            </button>

            <button class="action-btn" 
              title="Copy URL" 
              onclick="copyUrl('${item.url}', event)">
              📋
            </button>

            ${isCustom ? `
              <button class="action-btn delete-btn" 
                title="Delete Link" 
                onclick="deleteCustomLink('${item.id}', event)">
                🗑️
              </button>
            ` : ''}

            <span class="open-link-btn">
              Open ↗
            </span>
          </div>
        </a>
      `;
    }

    // Render Items (Supports Category Section Grouping!)
    function renderItems() {
      linksContainer.innerHTML = "";
      const allItems = getAllItems();

      const filtered = allItems.filter(item => {
        // Category Filter Match
        let categoryMatch = false;
        if (currentCategory === "all") {
          categoryMatch = true;
        } else if (currentCategory === "favorites") {
          categoryMatch = starredIds.includes(item.id);
        } else {
          categoryMatch = item.category === currentCategory;
        }

        // Search Match
        const q = searchQuery.toLowerCase();
        const searchMatch = !q || 
          item.title.toLowerCase().includes(q) || 
          item.url.toLowerCase().includes(q) || 
          (item.description && item.description.toLowerCase().includes(q));

        return categoryMatch && searchMatch;
      });

      visibleCount.textContent = filtered.length;
      activeFilterTag.textContent = `Filter: ${categoryNames[currentCategory] || 'Custom'}`;

      if (filtered.length === 0) {
        linksContainer.innerHTML = `
          <div style="text-align: center; padding: 60px 20px; border: 2px dashed var(--border-color); border-radius: var(--radius-lg); background: var(--bg-surface);">
            <div style="font-size: 44px; opacity: 0.5;">🔍</div>
            <div style="font-size: 18px; font-weight: 700; margin-top: 10px;">No links found</div>
            <div style="font-size: 13px; color: var(--text-muted); margin-top: 4px;">Try searching something else or switch categories.</div>
          </div>
        `;
        return;
      }

      // If viewing "All Items" without an active search query, display grouped by Category Sections!
      if (currentCategory === "all" && searchQuery === "") {
        const categoriesOrder = ["dashboards", "sheets", "docs", "apps", "tools"];

        categoriesOrder.forEach(catKey => {
          const catItems = filtered.filter(item => item.category === catKey);

          if (catItems.length > 0) {
            const sectionHtml = `
              <div class="category-group-section">
                <div class="category-header-title">
                  <span>${categoryConfig[catKey].icon} ${categoryConfig[catKey].title}</span>
                  <span class="cat-badge-count">${catItems.length} items</span>
                </div>
                <div class="${isListView ? 'links-list-mode' : 'links-grid'}">
                  ${catItems.map(item => createCardHtml(item)).join("")}
                </div>
              </div>
            `;
            linksContainer.insertAdjacentHTML("beforeend", sectionHtml);
          }
        });
      } else {
        // Direct Grid / List rendering for specific category / search results
        const containerHtml = `
          <div class="${isListView ? 'links-list-mode' : 'links-grid'}">
            ${filtered.map(item => createCardHtml(item)).join("")}
          </div>
        `;
        linksContainer.insertAdjacentHTML("beforeend", containerHtml);
      }
    }

    // Category Chip Click
    categoriesBar.addEventListener("click", (e) => {
      const chip = e.target.closest(".chip-btn");
      if (!chip) return;

      document.querySelectorAll(".chip-btn").forEach(btn => btn.classList.remove("active"));
      chip.classList.add("active");

      currentCategory = chip.dataset.category;
      updateCounts();
      renderItems();
    });

    // Select Category from Drawer
    function selectCategoryFromDrawer(categoryKey) {
      currentCategory = categoryKey;

      // Sync header chips
      document.querySelectorAll(".chip-btn").forEach(btn => {
        btn.classList.toggle("active", btn.dataset.category === categoryKey);
      });

      closeDrawer();
      updateCounts();
      renderItems();
    }

    // Search Input Event Listener
    searchInput.addEventListener("input", (e) => {
      searchQuery = e.target.value.trim();
      clearSearch.classList.toggle("visible", searchQuery.length > 0);
      renderItems();
    });

    clearSearch.addEventListener("click", () => {
      searchInput.value = "";
      searchQuery = "";
      clearSearch.classList.remove("visible");
      renderItems();
      searchInput.focus();
    });

    // Drawer Open / Close Logic
    function openDrawer() {
      mobileDrawer.classList.add("open");
      drawerBackdrop.classList.add("active");
    }

    function closeDrawer() {
      mobileDrawer.classList.remove("open");
      drawerBackdrop.classList.remove("active");
    }

    menuToggleBtn.addEventListener("click", openDrawer);
    closeDrawerBtn.addEventListener("click", closeDrawer);
    drawerBackdrop.addEventListener("click", closeDrawer);

    // Modal Open / Close Logic
    function openModal() {
      addModalOverlay.classList.add("active");
      document.getElementById("linkTitle").focus();
    }

    function closeModal() {
      addModalOverlay.classList.remove("active");
      addLinkForm.reset();
    }

    openAddModalBtn.addEventListener("click", openModal);
    drawerAddLinkBtn.addEventListener("click", () => {
      closeDrawer();
      openModal();
    });
    closeModalBtn.addEventListener("click", closeModal);
    cancelModalBtn.addEventListener("click", closeModal);

    // Code Help Modal
    helpCodeBtn.addEventListener("click", () => {
      closeDrawer();
      codeHelpOverlay.classList.add("active");
    });

    closeCodeHelpBtn.addEventListener("click", () => codeHelpOverlay.classList.remove("active"));
    closeHelpModalOk.addEventListener("click", () => codeHelpOverlay.classList.remove("active"));

    // Handle Custom Link Submission
    function handleCustomLinkSubmit(event) {
      event.preventDefault();

      const title = document.getElementById("linkTitle").value.trim();
      let url = document.getElementById("linkUrl").value.trim();
      const category = document.getElementById("linkCategory").value;
      let icon = document.getElementById("linkIcon").value.trim();

      if (!url.startsWith("http://") && !url.startsWith("https://")) {
        url = "https://" + url;
      }

      if (!icon) {
        icon = categoryConfig[category]?.icon || "🔗";
      }

      const newLinkObj = {
        id: "custom-" + Date.now(),
        title: title,
        url: url,
        category: category,
        icon: icon,
        isCustom: true,
        description: "User added custom link"
      };

      customLinks.push(newLinkObj);
      localStorage.setItem("omnidesk_custom_links", JSON.stringify(customLinks));

      closeModal();
      showToast("➕ New Custom Link added!");
      updateCounts();
      renderItems();
    }

    // View Switcher (Grid / List)
    viewToggle.addEventListener("click", () => {
      isListView = !isListView;
      viewIcon.textContent = isListView ? "📑" : "📱";
      showToast(isListView ? "Switched to List View" : "Switched to Grid View");
      renderItems();
    });

    // Theme Switcher (Dark / Light)
    themeToggle.addEventListener("click", () => {
      const currentTheme = document.documentElement.getAttribute("data-theme");
      const nextTheme = currentTheme === "dark" ? "light" : "dark";
      document.documentElement.setAttribute("data-theme", nextTheme);
      themeIcon.textContent = nextTheme === "dark" ? "🌙" : "☀️";
      localStorage.setItem("omnidesk_theme", nextTheme);
      showToast(`Switched to ${nextTheme.toUpperCase()} theme`);
    });

    const savedTheme = localStorage.getItem("omnidesk_theme");
    if (savedTheme) {
      document.documentElement.setAttribute("data-theme", savedTheme);
      themeIcon.textContent = savedTheme === "dark" ? "🌙" : "☀️";
    }

    // Global Keydown Shortcuts
    document.addEventListener("keydown", (e) => {
      if (e.key === "/" && document.activeElement !== searchInput) {
        e.preventDefault();
        searchInput.focus();
      } else if (e.key === "Escape") {
        closeDrawer();
        closeModal();
        codeHelpOverlay.classList.remove("active");
      }
    });

    // Initial Load
    updateCounts();
    renderItems();
  </script>
